while True:
    # Sube por la columna oeste censando

        spid.minar()

Mision 15 (Fase 8.5): PROTOCOLO MAESTRO SPIDO (El Bot Definitivo)
Terreno: Sector Abierto / Mapa Completo

def censar_y_minar():
    if spid.hay_mineral():
        spid.minar()
    # Si la tolva esta llena, se va solo a la nave, se vacia y continua
    if not spid.tiene_espacio():
        descargar_en_base()

def descargar_en_base():
    while not spid.en_base():
        spid.sur()
    spid.transferir()

def barrer_todo_el_mapa():
    ancho = spid.tamano_mundo()
    alto = spid.tamano_mundo() - 1

    # Recorre de izquierda a derecha (columna por columna)
    for col in range(ancho):
        # 1. Sube por la columna censando cada casilla
        for fila in range(alto):
            spid.norte()
            censar_y_minar()

        # 2. Salta al lado (este) si no es la ultima columna
        if col < ancho - 1:
            spid.este()
            censar_y_minar()

        # 3. Baja por la siguiente columna censando cada casilla
        for fila in range(alto):
            spid.sur()
            censar_y_minar()

    # Al terminar el mapa completo, entrega todo en la nave
    descargar_en_base()

# Bucle infinito del Rover 100% Autonomo
while True:
    barrer_todo_el_mapa()
